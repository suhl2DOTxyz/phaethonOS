# ==============================================================================
# PHAETHON OS - GRUB GRAPHICAL THEME (theme.txt)
# ==============================================================================
# Custom GRUB theme styled in the Zenless Zone Zero (ZZZ) visual style.
# Flat solid black background, centered golden star emblem, neon-lime tagline,
# and high-contrast neon-lime selection styling.
#
# ------------------------------------------------------------------------------
# SYNTAX -- GRUB uses TWO different separators, and mixing them silently kills
# the whole theme (GRUB aborts the parse and falls back to the plain text menu):
#
#   * Global properties, at the top level, use a COLON, and their values must
#     be quoted:                    desktop-color: "#0a0a0a"
#   * Component properties, inside a "+ component { }" block, use an EQUALS
#     sign:                         left = 50%-90
#
# Also: proportional values must not contain spaces. GRUB reads an unquoted
# expression up to the first whitespace, so "50% - 90" parses as "50%" and then
# chokes on the leftover "- 90". Write "50%-90".
# ------------------------------------------------------------------------------

# ------------------------------------------------------------------------------
# Global settings
# ------------------------------------------------------------------------------
title-text: ""

# Flat solid black. There is deliberately no `desktop-image` key: pointing it at
# an empty string makes GRUB try to load a bitmap called "", which fails, and a
# failed desktop-image aborts theme loading entirely.
desktop-color: "#0a0a0a"

# ------------------------------------------------------------------------------
# Centered distro golden star emblem
#
# `width`/`height` ON AN IMAGE ARE A FLOOR, NOT A CEILING. GRUB's canvas layout
# clamps every component up to its minimal size (gui_canvas.c):
#
#     comp->ops->get_minimal_size (comp, &mw, &mh);
#     if (w < (signed) mw) w = mw;
#
# and image_get_minimal_size returns the *raw bitmap's* dimensions. The source
# art was 251x251, so `width = 180` was ignored and the emblem drew at 251px --
# which is why it sat (251-180)/2 = 35.5px right of centre on the v1.0.0-Belle
# boot menu. `left = 50%-90` is only centred for a component that is genuinely
# 180 wide.
#
# logo.png is therefore generated at exactly 180x180 by tools/pixelate-logo.py
# from the master phaethon-logo.png. That makes the clamp a no-op, and it also
# means gui_image.c takes its `self->bitmap = self->raw_bitmap` fast path
# instead of resampling with GRUB_VIDEO_BITMAP_SCALE_METHOD_BEST -- so the
# 5px blocks baked into the file reach the screen 1:1 rather than being
# bilinearly smoothed back into a soft HD emblem that clashes with the 16px
# bitmap Unifont everything else on this screen is drawn in.
#
# If you change these numbers, re-run tools/pixelate-logo.py with a matching
# --size or the centring breaks again.
# ------------------------------------------------------------------------------
+ image {
    file = "logo.png"
    left = 50%-90
    top = 12%
    width = 180
    height = 180
}

# ------------------------------------------------------------------------------
# Distribution brand label
#
# `left = 0` + `width = 100%` + `align = "center"`: a label centres its text
# within its own box, so it has to span the screen for centring to mean
# anything. `left = 50%` would start the text at the middle instead.
# ------------------------------------------------------------------------------
+ label {
    text = "P H A E T H O N   O S"
    font = "Unifont Regular 16"
    color = "#c8ff00"
    left = 0
    top = 44%
    width = 100%
    align = "center"
}

# ------------------------------------------------------------------------------
# Distro brand tagline/motto
# ------------------------------------------------------------------------------
+ label {
    text = "Though greatly he failed, more greatly he dared."
    font = "Unifont Regular 16"
    color = "#ffffff"
    left = 0
    top = 50%
    width = 100%
    align = "center"
}

# ------------------------------------------------------------------------------
# Graphical boot list selection menu
#
# The highlight is a nine-slice box: `selected_item_pixmap_style` takes a
# pattern and GRUB substitutes c/n/s/e/w/nw/ne/sw/se for the "*", loading
# select_c.png, select_n.png and so on. (The property is
# `selected_item_pixmap_style`, not `selected_item_pixmap` -- the latter is not
# a real property and is silently ignored.) All nine slices are the same flat
# lime, so they fuse into one solid highlight bar with black text on it.
#
# Both fonts are "Unifont Regular 16", the single face inside the shipped
# unicode.pf2. There is no "Unifont Bold 16"; naming a font GRUB does not have
# just falls back to the built-in one, so the selected row would stop matching
# the rest of the menu.
# ------------------------------------------------------------------------------
+ boot_menu {
    left = 50%-250
    top = 58%
    width = 500
    height = 30%

    item_font = "Unifont Regular 16"
    selected_item_font = "Unifont Regular 16"

    item_color = "#ffffff"
    selected_item_color = "#0a0a0a"
    selected_item_pixmap_style = "select_*.png"

    item_height = 28
    item_spacing = 8
    item_padding = 8

    scrollbar = false
}

# ------------------------------------------------------------------------------
# Countdown to the default entry.
#
# The id "__timeout__" is what makes GRUB drive this component with the
# remaining seconds; without it, it is just a static bar.
# ------------------------------------------------------------------------------
+ progress_bar {
    id = "__timeout__"
    left = 50%-250
    top = 90%
    width = 500
    height = 20

    font = "Unifont Regular 16"
    text = "@TIMEOUT_NOTIFICATION_MIDDLE@"
    text_color = "#888888"

    fg_color = "#c8ff00"
    bg_color = "#1a1a1a"
    border_color = "#333333"
}
